cmake_minimum_required(VERSION 3.22)
project(foresight_speech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(nlohmann_json QUIET)
if(NOT nlohmann_json_FOUND)
  include(FetchContent)
  FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 65ee68451d8eb2b5f3a30b410476ab83deb3289b)
  FetchContent_MakeAvailable(json)
endif()
find_package(Threads REQUIRED)
if(NOT FORESIGHT_WHISPER_SOURCE)
  message(FATAL_ERROR "FORESIGHT_WHISPER_SOURCE must name the pinned whisper.cpp checkout")
endif()
add_subdirectory("${FORESIGHT_WHISPER_SOURCE}" whisper)
add_executable(foresight-whisper-worker worker.cpp)
target_link_libraries(foresight-whisper-worker PRIVATE whisper nlohmann_json::nlohmann_json Threads::Threads)
set_target_properties(foresight-whisper-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
