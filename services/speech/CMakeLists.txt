cmake_minimum_required(VERSION 3.22)
project(foresight_speech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra)
find_package(nlohmann_json QUIET)
if(NOT TARGET nlohmann_json::nlohmann_json)
  include(FetchContent)
  set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG 65ee68451d8eb2b5f3a30b410476ab83deb3289b GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(nlohmann_json)
endif()
find_package(Threads REQUIRED)
if(FORESIGHT_WHISPER_SOURCE)
  # Developer builds can still build the pinned source checkout directly.
  add_subdirectory("${FORESIGHT_WHISPER_SOURCE}" whisper)
else()
  # CI workers link the installed, separately cached vendor libraries.
  find_package(whisper CONFIG REQUIRED)
endif()
add_executable(foresight-whisper-worker worker.cpp)
target_link_libraries(foresight-whisper-worker PRIVATE whisper nlohmann_json::nlohmann_json Threads::Threads)
set_target_properties(foresight-whisper-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
