cmake_minimum_required(VERSION 3.20)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/OpenForesightVersion.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/MacOSDeploymentTarget.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/ForesightTestHooks.cmake")
project(foresight_helper VERSION ${OPEN_FORESIGHT_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra)
find_package(Protobuf CONFIG REQUIRED)
find_package(gRPC CONFIG REQUIRED)
find_package(SQLite3 REQUIRED)
# CMake before 4.3 exports the legacy name (including Ubuntu 24.04's 3.28).
if(NOT TARGET SQLite3::SQLite3)
 add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
endif()
find_package(nlohmann_json REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RECOVERED_STEM "rag_service")
set(GENERATED "${GEN}/${RECOVERED_STEM}.pb.cc" "${GEN}/${RECOVERED_STEM}.grpc.pb.cc" "${GEN}/helper_control.pb.cc" "${GEN}/helper_control.grpc.pb.cc")
add_custom_command(OUTPUT "${GEN}/${RECOVERED_STEM}.pb.cc" "${GEN}/${RECOVERED_STEM}.pb.h" "${GEN}/${RECOVERED_STEM}.grpc.pb.cc" "${GEN}/${RECOVERED_STEM}.grpc.pb.h"
 COMMAND ${Python3_EXECUTABLE} "${REPO_ROOT}/tools/generate_bindings.py" --language cpp --output ${GEN} --protoc $<TARGET_FILE:protobuf::protoc> --grpc-plugin $<TARGET_FILE:gRPC::grpc_cpp_plugin>
 DEPENDS "${REPO_ROOT}/proto/rag/rag_service.descriptor.pb" "${REPO_ROOT}/proto/rag/rag_service.proto" "${REPO_ROOT}/tools/generate_bindings.py" VERBATIM)
add_custom_command(OUTPUT "${GEN}/helper_control.pb.cc" "${GEN}/helper_control.pb.h" "${GEN}/helper_control.grpc.pb.cc" "${GEN}/helper_control.grpc.pb.h"
 COMMAND protobuf::protoc ARGS --proto_path=${REPO_ROOT}/proto/linux --cpp_out=${GEN} --grpc_out=${GEN} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> ${REPO_ROOT}/proto/linux/helper_control.proto
 DEPENDS "${REPO_ROOT}/proto/linux/helper_control.proto" VERBATIM)
add_library(foresight_protocol ${GENERATED})
target_include_directories(foresight_protocol PUBLIC "${GEN}")
target_link_libraries(foresight_protocol PUBLIC protobuf::libprotobuf gRPC::grpc++)
add_executable(ForesightHelper main.cpp)
target_compile_definitions(ForesightHelper PRIVATE FORESIGHT_ENABLE_TEST_HOOKS=${FORESIGHT_PRODUCTION_TEST_HOOKS})
target_compile_options(ForesightHelper PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ForesightHelper PRIVATE foresight_protocol SQLite3::SQLite3 nlohmann_json::nlohmann_json)
set(FORESIGHT_BUILD_REVISION "" CACHE STRING "Source revision for exported source builds")
include("${REPO_ROOT}/cmake/ForesightBuildIdentity.cmake")
foresight_build_revision(BUILD_REV "${REPO_ROOT}")
foresight_build_identity_source(IDENTITY_SOURCE
 "${CMAKE_CURRENT_SOURCE_DIR}/build_identity.cpp.in" "${CMAKE_CURRENT_BINARY_DIR}")
target_sources(ForesightHelper PRIVATE "${IDENTITY_SOURCE}")
install(TARGETS ForesightHelper RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
 find_package(Threads REQUIRED)
 add_executable(helper_reconfig_wait_test "${REPO_ROOT}/tests/helper/reconfig_wait.cpp")
 target_include_directories(helper_reconfig_wait_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(helper_reconfig_wait_test PRIVATE Threads::Threads)
 add_test(NAME helper_reconfig_wait COMMAND helper_reconfig_wait_test)
 add_executable(helper_vector_index_test "${REPO_ROOT}/tests/helper/vector_index.cpp")
 target_include_directories(helper_vector_index_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(helper_vector_index_test PRIVATE Threads::Threads)
 add_test(NAME helper_vector_index COMMAND helper_vector_index_test)
 add_test(NAME helper_vector_index_require COMMAND helper_vector_index_test --require-failure)
 set_tests_properties(helper_vector_index_require PROPERTIES WILL_FAIL TRUE)
 set(FORESIGHT_CONTRACT_HELPER ForesightHelper)
 string(TOLOWER "${CMAKE_BUILD_TYPE}" _foresight_helper_configuration)
 if(CMAKE_CONFIGURATION_TYPES OR NOT FORESIGHT_ENABLE_TEST_HOOKS OR _foresight_helper_configuration MATCHES "^(release|relwithdebinfo|minsizerel)$")
  # The release executable remains hook-free while crash-consistency contracts
  # exercise an explicitly instrumented test-only executable.
  add_executable(ForesightHelperTest main.cpp)
  target_sources(ForesightHelperTest PRIVATE "${IDENTITY_SOURCE}")
  target_compile_definitions(ForesightHelperTest PRIVATE FORESIGHT_ENABLE_TEST_HOOKS=1)
  target_compile_options(ForesightHelperTest PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(ForesightHelperTest PRIVATE foresight_protocol SQLite3::SQLite3 nlohmann_json::nlohmann_json)
  set(FORESIGHT_CONTRACT_HELPER ForesightHelperTest)
 endif()
 add_test(NAME helper_release_hooks COMMAND ${Python3_EXECUTABLE} "${REPO_ROOT}/tests/helper/release_hooks.py" $<TARGET_FILE:ForesightHelper> --expected-enabled ${FORESIGHT_PRODUCTION_TEST_HOOKS})
 add_executable(helper_contract_test "${REPO_ROOT}/tests/helper/contract.cpp")
 target_link_libraries(helper_contract_test PRIVATE foresight_protocol SQLite3::SQLite3)
 add_test(NAME helper_contract COMMAND helper_contract_test $<TARGET_FILE:${FORESIGHT_CONTRACT_HELPER}>)
endif()
